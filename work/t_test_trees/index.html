# Categorise the trees
unhealthy_cols = ['0U1 - Unhealthy', 'A1 - Unhealthy', 'A2 - Unhealthy']
healthy_cols = ['B1 - Healthy', 'B2 - Healthy', 'OH1 - Healthy']

# Abundance = Row Averages
  # The average count of each family across (un)healthy trees
trees['abun_unhealthy'] = trees[unhealthy_cols].mean(axis=1)
trees['abun_healthy'] = trees[healthy_cols].mean(axis=1)

# Diversity = Normalised scale of spread
  # Basically a value between 0-1
  # 1 = There are counts >0 across everything
  # 0 = There are NO counts >0 across everything
trees['dive_unhealthy'] = (
    trees[unhealthy_cols] > 0
    ).sum(axis=1) / len(unhealthy_cols)
trees['dive_healthy'] = (
    trees[healthy_cols] > 0
    ).sum(axis=1) / len(healthy_cols)

# Set hierarchy of classification taxonomy
group_cols = ['Kingdom', 'Phylum', 'Class', 'Order', 'Family']

# Pivot the data
# Abundance is based on row averages
pivot_trees = (
    trees.groupby(group_cols, as_index=False)
      .agg({
          'abun_unhealthy': 'mean',
          'abun_healthy': 'mean',
          'dive_unhealthy': 'mean',
          'dive_healthy': 'mean'
      })
)

The Hypothesis¶

$H_0$ = "There is no difference in abundance or diversity between healthy and unhealthy trees."

$H_1$ = "Unhealthy trees contain higher abundance and lower diversity."

Hypothesis Test: Paired t-test¶

# Importing a Python library that can conduct t-testing
from scipy.stats import ttest_rel

abun_unhealthy = pivot_trees['abun_unhealthy']
abun_healthy = pivot_trees['abun_healthy']

dive_unhealthy = pivot_trees['dive_unhealthy']
dive_healthy = pivot_trees['dive_healthy']

# Abundance test (H1: adundance unhealthy > abundance healthy)
t_abun = ttest_rel(abun_unhealthy, abun_healthy, alternative='greater')

# Diversity test (H1: diversity unhealthy < diversity healthy)
t_dive = ttest_rel(dive_unhealthy, dive_healthy, alternative='less')

results = pd.DataFrame({
    "Metric": ["Abundance", "Diversity"],
    "Comparison": ["Unhealthy vs Healthy", "Unhealthy vs Healthy"],
    "Hypothesis": ["Unhealthy > Healthy", "Unhealthy < Healthy"],
    "t-statistic": [round(t_abun.statistic,4), round(t_dive.statistic,4)],
    "p-value": [round(t_abun.pvalue,4), round(t_dive.pvalue,4)],
    "Conclusion": ["Not significant", "Not significant"]
})
